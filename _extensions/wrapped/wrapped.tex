% wrapped: the wrappedfigure environment, a key-value front end for wrapfig's
% wrapfigure. The filter writes
% \begin{wrappedfigure}[side=right,width=0.4\linewidth,gap=1em,lines=12].
%
% The sides are wrapfig's floating placements, R and L: a figure that does
% not fit on the page moves to the start of a paragraph on the next page.
%
% wrapfig puts \columnsep between the figure and the text. It reads
% \columnsep only in \WF@putfigmaybe, when it places the figure at the start
% of a later paragraph, inside a group that \WF@floathand opens. So the
% environment keeps its gap in \wrapped@gap when it ends, and
% \WF@putfigmaybe sets \columnsep to it inside that group. The document's
% \columnsep never changes. \WF@startwrapping and \WFclear drop the kept gap
% once the figure is placed or put out without text.
%
% The filter writes \wrappedstop after the paragraphs beside a figure, at
% the block that stops the wrap, so that block starts below the figure.
% wrapfig holds one figure at a time, so a wrappedfigure also ends an
% earlier wrap first, in the same way. A figure that moved and still waits
% for a paragraph is put out there, centered, with no text beside it. A wrap
% whose text is still narrowed ends: the lines that the figure's height
% still needs are skipped, and \WFclear ends the wrap. In a wrappedfigure,
% \WFclear ends it inside the environment's group, so \wrapped@afterenv
% ends it again after the group.
%
% Where the text runs in columns, inside multicols or where \if@twocolumn
% is true, a wrappedfigure does not wrap: it floats as a normal figure, or
% it is set in place inside multicols or when it has no caption. LaTeX
% writes a warning to its log: once for two-column text, and once for each
% figure in multicols.
\usepackage{wrapfig}
\makeatletter
\def\wrapped@gap{\columnsep}
\newcommand*\wrapped@setgap[1]{\xdef\wrapped@gap{\the\dimexpr #1\relax}}
\newcommand*\wrapped@cleargap{\gdef\wrapped@gap{\columnsep}}
\let\wrapped@putfigmaybe\WF@putfigmaybe
\def\WF@putfigmaybe{\columnsep=\wrapped@gap\relax \wrapped@putfigmaybe}
\g@addto@macro\WF@startwrapping{\wrapped@cleargap}
% \WFclear as in wrapfig, but a figure that waits is centered. LaTeX runs
% \WFclear at each \clearpage, the end of the document included.
\renewcommand\WFclear{\par
  \ifvoid\WF@box\else
    \vskip\intextsep
    \hbox to\linewidth{\hss\box\WF@box\hss}%
    \vskip\intextsep
    \let\everypar\WF@@everypar \everypar\expandafter{\the\WF@everypar}%
  \fi
  \global\c@WF@wrappedlines\z@ \WF@finale
  \wrapped@cleargap}
% wrapfig counts the lines beside a figure from \WF@size, the figure's
% height, or takes the wrap-lines value. \wrapped@startlines keeps the count
% that the wrap started with, and \wrapped@startbs the \baselineskip that
% wrapfig counted with.
\newcount\wrapped@startlines
\newdimen\wrapped@startbs
\g@addto@macro\WF@startwrapping{%
  \global\wrapped@startlines\c@WF@wrappedlines
  \global\wrapped@startbs\baselineskip}
% The lines skipped are the lines that wrapfig counts from the height, less
% the lines already set beside the figure, less one. A wrap-lines value
% larger than the height needs adds no space.
\newcommand*\wrapped@endwrap{%
  \ifvoid\WF@box\else \WFclear \fi
  \ifx\parshape\WF@fudgeparshape
    \par
    \@tempdima\WF@size \advance\@tempdima1.1\wrapped@startbs
    \divide\@tempdima\wrapped@startbs \@tempcnta\@tempdima
    \advance\@tempcnta-\wrapped@startlines
    \advance\@tempcnta\c@WF@wrappedlines \advance\@tempcnta\m@ne
    \ifnum\@tempcnta>\z@ \vspace{\@tempcnta\wrapped@startbs}\fi
    \WFclear
  \fi}
\newcommand*\wrappedstop{\wrapped@endwrap}
% wrapfig runs \WF@startfloating after the environment's groups end. Before
% it, \wrapped@afterenv ends an earlier wrap there and records that group
% level for \WF@floathand below.
\def\wrapped@level{0}
\newcommand*\wrapped@afterenv{%
  \xdef\wrapped@level{\the\currentgrouplevel}%
  \ifx\parshape\WF@fudgeparshape \WF@finale \fi}
\let\wrapped@startfloating\WF@startfloating
\def\WF@startfloating{\wrapped@afterenv\wrapped@startfloating}
% A paragraph inside a group, such as a heading or the end of a callout,
% can start where a moved figure waits. A figure that wrapfig placed there
% would lose its wrap when the group ends, and the text after it would run
% over the figure. So wrapfig places a moved figure only at a paragraph on
% the group level where its environment ended.
\let\wrapped@floathand\WF@floathand
\def\WF@floathand{%
  \ifnum\currentgrouplevel=\wrapped@level\relax
    \expandafter\wrapped@floathand
  \fi}
% Where the text runs in columns, a wrappedfigure does not wrap. The
% hooks mark the inside of a multicols environment, where a float is lost,
% and the mark ends with the environment's group. \if@twocolumn stays true
% inside multicols in two-column text, so the mark is tested first.
\newif\ifwrapped@multicols
\AddToHook{env/multicols/begin}{\wrapped@multicolstrue}
\AddToHook{env/multicols*/begin}{\wrapped@multicolstrue}
\newif\ifwrapped@twowarned
\newcommand*\wrapped@warnmulticols{%
  \PackageWarning{wrapped}{Figure in multicols set as normal figure}}
\newcommand*\wrapped@warntwocolumn{%
  \ifwrapped@twowarned\else
    \global\wrapped@twowarnedtrue
    \PackageWarningNoLine{wrapped}{Two-column text: figures set as normal figures}%
  \fi}
\makeatother
\ExplSyntaxOn
\tl_new:N \l__wrapped_side_tl
\tl_new:N \l__wrapped_mode_tl
\bool_new:N \l__wrapped_caption_bool
\keys_define:nn { wrapped }
  {
    side .choice:,
    side / left .code:n = \tl_set:Nn \l__wrapped_side_tl { L },
    side / right .code:n = \tl_set:Nn \l__wrapped_side_tl { R },
    side .initial:n = right,
    width .tl_set:N = \l__wrapped_width_tl,
    width .initial:n = 0.4\linewidth,
    gap .tl_set:N = \l__wrapped_gap_tl,
    gap .initial:n = 1em,
    lines .tl_set:N = \l__wrapped_lines_tl,
    lines .initial:n = ,
    caption .bool_set:N = \l__wrapped_caption_bool,
    caption .initial:n = true,
  }
% The mode of a wrappedfigure: w wraps, f is a figure float in two-column
% text, and p sets the figure in place, inside multicols or for an image
% with no caption in two-column text.
\cs_new_protected:Npn \__wrapped_set_mode:
  {
    \legacy_if:nTF { wrapped@multicols }
      {
        \tl_set:Nn \l__wrapped_mode_tl { p }
        \use:c { wrapped@warnmulticols }
      }
      {
        \legacy_if:nTF { @twocolumn }
          {
            \bool_if:NTF \l__wrapped_caption_bool
              { \tl_set:Nn \l__wrapped_mode_tl { f } }
              { \tl_set:Nn \l__wrapped_mode_tl { p } }
            \use:c { wrapped@warntwocolumn }
          }
          { \tl_set:Nn \l__wrapped_mode_tl { w } }
      }
  }
\NewDocumentEnvironment { wrappedfigure } { O{} }
  {
    \keys_set:nn { wrapped } { #1 }
    \use:c { wrapped@endwrap }
    \__wrapped_set_mode:
    \exp_args:NV \str_case:nn \l__wrapped_mode_tl
      {
        { w }
          {
            % A caption in a narrow box reads better set plain and
            % ragged-right.
            \cs_if_exist:NT \captionsetup
              { \captionsetup { format = plain, justification = raggedright } }
            \use:e
              {
                \exp_not:N \wrapfigure
                \tl_if_empty:NF \l__wrapped_lines_tl
                  { [ \exp_not:V \l__wrapped_lines_tl ] }
                { \exp_not:V \l__wrapped_side_tl }
                { \exp_not:V \l__wrapped_width_tl }
              }
          }
        { f }
          {
            \begin { figure }
            \centering
            \begin { minipage } { \l__wrapped_width_tl }
          }
        { p }
          {
            \begin { center }
            \begin { minipage } { \l__wrapped_width_tl }
            \cs_set:cpn { @captype } { figure }
          }
      }
  }
  {
    \exp_args:NV \str_case:nn \l__wrapped_mode_tl
      {
        { w }
          {
            \endwrapfigure
            \use:c { wrapped@setgap } { \l__wrapped_gap_tl }
          }
        { f } { \end { minipage } \end { figure } }
        { p } { \end { minipage } \end { center } }
      }
  }
\ExplSyntaxOff
